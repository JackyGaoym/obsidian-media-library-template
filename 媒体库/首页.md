const items = dv.pages('"媒体库/作品"')
  .where(page => page.note_type === "media")
  .array();

const typeMeta = {
  book: { label: "图书", unit: "本", path: "媒体库/图书" },
  tv: { label: "电视剧", unit: "部", path: "媒体库/电视剧" },
  movie: { label: "电影", unit: "部", path: "媒体库/电影" },
  anime: { label: "动漫", unit: "部", path: "媒体库/动漫" },
  game: { label: "游戏", unit: "款", path: "媒体库/游戏" }
};

const timestamp = value => {
  if (!value) return 0;
  if (typeof value.toMillis === "function") return value.toMillis();
  const parsed = Date.parse(String(value));
  return Number.isNaN(parsed) ? 0 : parsed;
};

const mediaTitle = page => page.title || page.file.name;

const coverUrl = page => {
  const raw = page.cover?.path ?? String(page.cover || "");
  const linkPath = raw.replace(/^\[\[/, "").replace(/\]\]$/, "").split("|")[0];
  if (!linkPath) return "";
  const file = app.metadataCache.getFirstLinkpathDest(linkPath, page.file.path);
  return file ? app.vault.getResourcePath(file) : "";
};

const addInternalLink = (parent, path, label, cls = "") => {
  const link = parent.createEl("a", { cls: `internal-link ${cls}`.trim(), text: label });
  link.setAttr("data-href", path);
  link.setAttr("href", /\.(md|base)$/.test(path) ? path : `${path}.md`);
  return link;
};

const numberFrom = value => {
  if (typeof value === "number") return Number.isFinite(value) ? value : 0;
  const match = String(value ?? "").replaceAll(",", "").match(/\d+(?:\.\d+)?/);
  return match ? Number(match[0]) : 0;
};

const progressInputNumber = value => {
  const normalized = String(value ?? "").trim().replaceAll(",", "");
  return /^\d+(?:\.\d+)?$/.test(normalized) ? Number(normalized) : null;
};

const progressFor = page => {
  const type = page.media_type;
  const config = type === "book"
    ? { field: "current_page", totalField: "page_count", unit: "页", step: 1 }
    : (type === "tv" || type === "anime")
      ? { field: "current_episode", totalField: "episode_count", unit: "集", step: 1 }
      : type === "movie"
        ? { field: "current_minutes", totalField: "runtime_minutes", unit: "分", step: 10 }
        : type === "game"
          ? { field: "progress_percent", totalField: null, unit: "%", step: 5, fixedTotal: 100 }
          : null;

  if (!config) return null;
  const total = config.fixedTotal ?? Math.max(0, Math.round(numberFrom(page[config.totalField])));
  const current = Math.max(0, Math.round(numberFrom(page[config.field])));
  const known = total > 0;
  return {
    ...config,
    current: known ? Math.min(current, total) : current,
    total,
    known
  };
};

const progressLabel = state => state.unit === "%"
  ? `${state.current}%`
  : `${state.current} / ${state.known ? state.total : "?"} ${state.unit}`;

const progressPercent = state => state.known
  ? Math.max(0, Math.min(100, state.current / state.total * 100))
  : 0;

const setAppIcon = (element, name, fallback) => {
  element.empty();
  try {
    const iconSetter = typeof setIcon === "function"
      ? setIcon
      : (typeof require === "function" ? require("obsidian").setIcon : null);
    if (iconSetter) {
      iconSetter(element, name);
      return;
    }
  } catch (error) {
  }
  element.setText(fallback);
};

const showProgressNotice = message => {
  try {
    const NoticeClass = typeof require === "function" ? require("obsidian").Notice : null;
    if (NoticeClass) new NoticeClass(message);
  } catch (error) {
    console.error(message, error);
  }
};

const updateFrontmatter = async (page, values) => {
  const file = app.vault.getAbstractFileByPath(page.file.path);
  if (!file) throw new Error(`找不到文件：${page.file.path}`);
  await app.fileManager.processFrontMatter(file, frontmatter => {
    for (const [key, value] of Object.entries(values)) frontmatter[key] = value;
  });
};

const root = dv.container.createDiv({ cls: "media-home-dashboard" });
const latestUpdate = items.reduce((latest, page) =>
  Math.max(latest, timestamp(page.file?.mtime), timestamp(page.added_at)), 0);
const latestUpdateLabel = latestUpdate
  ? (() => {
      const date = new Date(latestUpdate);
      return `${date.getMonth() + 1} 月 ${date.getDate()} 日更新`;
    })()
  : "等待第一件收藏";

// 顶部标题与主要动作
const header = root.createDiv({ cls: "media-home-header" });
const identity = header.createDiv({ cls: "media-home-identity" });
identity.createEl("h1", { text: "媒体库" });
identity.createSpan({
  cls: "media-home-subtitle",
  text: `我的收藏持续生长 · ${items.length} 部作品 · ${latestUpdateLabel}`
});

const headerActions = header.createDiv({ cls: "media-home-actions" });
const primaryActions = headerActions.createDiv({ cls: "media-home-primary-actions" });
const addButton = primaryActions.createEl("button", { cls: "media-home-main-action" });
addButton.createSpan({ cls: "media-home-action-icon", text: "＋" });
addButton.createSpan({ text: "新增作品" });
addButton.addEventListener("click", () => {
  app.commands.executeCommandById("quickadd:choice:a8f3f962-2f97-4b42-86fb-8c8e3571d9e0");
});
const collectionButton = primaryActions.createEl("button", { cls: "media-home-main-action" });
collectionButton.createSpan({ text: "合集与系列" });
collectionButton.addEventListener("click", () => {
  app.workspace.openLinkText("媒体库/合集索引", "", false);
});
const utilities = headerActions.createDiv({ cls: "media-home-utilities" });
addInternalLink(utilities, "媒体库/使用说明", "使用说明");

// 统计与分类共用一条紧凑信息带
const rated = items.filter(page => typeof page.rating === "number" && page.rating > 0);
const average = rated.length
  ? (rated.reduce((sum, page) => sum + page.rating, 0) / rated.length).toFixed(1)
  : "—";
const stats = [
  ["全部作品", items.length],
  ["进行中", items.filter(page => page.status === "进行中").length],
  ["已完成", items.filter(page => page.status === "已完成").length],
  ["平均评分", average]
];

const summary = root.createDiv({ cls: "media-home-summary" });
const statsWrap = summary.createDiv({ cls: "media-home-stats" });
for (const [label, value] of stats) {
  const stat = statsWrap.createDiv({ cls: "media-home-stat" });
  stat.createDiv({ cls: "media-home-stat-value", text: String(value) });
  stat.createDiv({ cls: "media-home-stat-label", text: label });
}

const categories = summary.createDiv({ cls: "media-home-categories" });
for (const [type, meta] of Object.entries(typeMeta)) {
  const count = items.filter(page => page.media_type === type).length;
  const link = addInternalLink(categories, meta.path, "", `media-home-category is-${type}`);
  link.createSpan({ cls: "media-home-category-name", text: meta.label });
  link.createSpan({ cls: "media-home-category-count", text: `${count} ${meta.unit}` });
}

// 主展示区：左侧最近加入，右侧当前进度
const content = root.createDiv({ cls: "media-home-content" });
const recentSection = content.createDiv({ cls: "media-home-section media-home-recent" });
const recentHeading = recentSection.createDiv({ cls: "media-home-section-heading" });
recentHeading.createEl("h2", { text: "最近加入" });
addInternalLink(recentHeading, "媒体库/视图/全部作品.base", "查看全部", "media-home-more");

const recent = [...items]
  .sort((a, b) => timestamp(b.added_at) - timestamp(a.added_at)
    || timestamp(b.file.mtime) - timestamp(a.file.mtime))
  .slice(0, 10);
const recentGrid = recentSection.createDiv({ cls: "media-home-poster-grid" });

if (recent.length === 0) {
  recentGrid.createDiv({
    cls: "media-home-empty",
    text: "媒体库还是空的。点击上方的「新增作品」，从第一部作品开始。"
  });
}

for (const page of recent) {
  const card = recentGrid.createEl("a", { cls: "media-home-poster internal-link" });
  card.setAttr("data-href", page.file.path);
  card.setAttr("href", page.file.path);
  card.setAttr("aria-label", `打开 ${mediaTitle(page)}`);

  const visual = card.createDiv({ cls: `media-home-poster-visual is-${page.media_type || "media"}` });
  const cover = coverUrl(page);
  if (cover) {
    const image = visual.createEl("img");
    image.setAttr("src", cover);
    image.setAttr("alt", `${mediaTitle(page)}封面`);
    image.setAttr("loading", "lazy");
  } else {
    visual.createSpan({ cls: "media-home-cover-fallback", text: typeMeta[page.media_type]?.label || "作品" });
  }

  card.createDiv({ cls: "media-home-poster-title", text: mediaTitle(page) });
  const meta = card.createDiv({ cls: "media-home-poster-meta" });
  meta.createSpan({
    cls: `media-home-type is-${page.media_type || "media"}`,

  });
  meta.createSpan({ cls: `media-home-status is-${page.status || "unknown"}`, text: page.status || "未设置" });
  if (typeof page.rating === "number" && page.rating > 0) {
    meta.createSpan({ cls: "media-home-rating", text: `${page.rating} ★` });
  }
}

const activeSection = content.createDiv({ cls: "media-home-section media-home-active" });
const activeHeading = activeSection.createDiv({ cls: "media-home-section-heading" });
activeHeading.createEl("h2", { text: "正在进行" });

const activeItems = items
  .filter(page => page.status === "进行中")
  .sort((a, b) => timestamp(b.file?.mtime) - timestamp(a.file?.mtime)
    || timestamp(b.added_at) - timestamp(a.added_at));
const activeList = activeSection.createDiv({ cls: "media-home-active-list" });

if (activeItems.length === 0) {
  activeList.createDiv({ cls: "media-home-empty", text: "当前没有进行中的作品。" });
}

for (const page of activeItems) {
  const row = activeList.createDiv({
    cls: `media-home-active-item is-${page.media_type || "media"}`,
    attr: { role: "link", tabindex: "0", "aria-label": `打开 ${mediaTitle(page)}` }
  });
  const openPage = () => app.workspace.openLinkText(page.file.path, "", false);
  row.addEventListener("click", event => {
    if (!event.target.closest("button")) openPage();
  });
  row.addEventListener("keydown", event => {
    if (event.target !== row || (event.key !== "Enter" && event.key !== " ")) return;
    event.preventDefault();
    openPage();
  });

  const thumb = row.createDiv({ cls: `media-home-active-cover is-${page.media_type || "media"}` });
  const cover = coverUrl(page);
  if (cover) {
    const image = thumb.createEl("img");
    image.setAttr("src", cover);
    image.setAttr("alt", "");
    image.setAttr("loading", "lazy");
  } else {
    thumb.createSpan({ text: typeMeta[page.media_type]?.label || "作品" });
  }

  const copy = row.createDiv({ cls: "media-home-active-copy" });
  copy.createDiv({ cls: "media-home-active-title", text: mediaTitle(page) });
  const detail = copy.createDiv({ cls: "media-home-active-detail" });
  detail.createSpan({
    cls: `media-home-active-type is-${page.media_type || "media"}`,
    text: typeMeta[page.media_type]?.label || "作品"
  });
  if (typeof page.rating === "number" && page.rating > 0) {
    detail.createSpan({ text: `${page.rating} ★` });
  }

  const state = progressFor(page);
  if (!state) continue;

  const progressRow = copy.createDiv({ cls: "media-home-progress-row" });
  const track = progressRow.createDiv({ cls: "media-home-progress-track" });
  const fill = track.createDiv({ cls: "media-home-progress-fill" });
  const controls = progressRow.createDiv({ cls: "media-home-progress-controls" });
  const decrease = controls.createEl("button", {
    cls: "media-home-progress-button",
    attr: { type: "button", "aria-label": `减少《${mediaTitle(page)}》的进度` }
  });
  setAppIcon(decrease, "minus", "−");
  const valueButton = controls.createEl("button", {
    cls: "media-home-progress-value",
    attr: { type: "button", "aria-label": `编辑《${mediaTitle(page)}》的进度` }
  });
  const action = controls.createEl("button", {
    cls: "media-home-progress-button media-home-progress-action",
    attr: { type: "button" }
  });

  const isComplete = () => state.known && state.current >= state.total;
  const syncControls = () => {
    const complete = isComplete();
    track.toggleClass("is-unknown", !state.known);
    fill.style.width = `${progressPercent(state)}%`;
    valueButton.setText(progressLabel(state));
    decrease.disabled = state.current <= 0;
    action.toggleClass("is-complete", complete);
    action.setAttr("aria-label", complete
      ? `将《${mediaTitle(page)}》标记为已完成`
      : `增加《${mediaTitle(page)}》的进度`);
    if (complete) action.setText("完成");
    else setAppIcon(action, "plus", "+");
  };

  const setBusy = busy => {
    row.toggleClass("is-updating", busy);
    decrease.disabled = busy || state.current <= 0;
    valueButton.disabled = busy;
    action.disabled = busy;
  };

  const saveProgress = async values => {
    setBusy(true);
    try {
      await updateFrontmatter(page, values);
      return true;
    } catch (error) {
      console.error(error);
      showProgressNotice("进度更新失败，请稍后重试。");
      return false;
    } finally {
      setBusy(false);
    }
  };

  decrease.addEventListener("click", async event => {
    event.preventDefault();
    event.stopPropagation();
    const next = Math.max(0, state.current - state.step);
    if (next === state.current || !await saveProgress({ [state.field]: next })) return;
    state.current = next;
    syncControls();
  });

  valueButton.addEventListener("click", async event => {
    event.preventDefault();
    event.stopPropagation();
    const example = state.unit === "%"
      ? "例如：65"
      : `可以输入当前${state.unit}数，或输入“当前/总${state.unit}数”`;
    const input = window.prompt(`编辑《${mediaTitle(page)}》进度\n${example}`, state.known
      ? `${state.current}/${state.total}`
      : String(state.current));
    if (input === null) return;

    const rawParts = input.split(/[\/／]/);
    const parts = rawParts.map(progressInputNumber);
    if (parts.length > 2 || parts.some(value => value === null)) {
      showProgressNotice("请输入有效的进度数字。");
      return;
    }
    const nextCurrent = Math.max(0, Math.round(parts[0]));

    const values = { [state.field]: nextCurrent };
    let nextTotal = state.total;
    if (state.totalField && parts.length > 1) {
      nextTotal = Math.max(0, Math.round(parts[1]));
      values[state.totalField] = nextTotal;
    }
    const nextKnown = state.fixedTotal ? true : nextTotal > 0;
    const clampedCurrent = nextKnown ? Math.min(nextCurrent, nextTotal) : nextCurrent;
    values[state.field] = clampedCurrent;
    if (!await saveProgress(values)) return;
    state.current = clampedCurrent;
    state.total = nextTotal;
    state.known = nextKnown;
    syncControls();
  });

  action.addEventListener("click", async event => {
    event.preventDefault();
    event.stopPropagation();
    if (isComplete()) {
      if (!await saveProgress({ status: "已完成" })) return;
      row.addClass("is-finished");
      window.setTimeout(() => {
        row.remove();
        if (!activeList.querySelector(".media-home-active-item")) {
          activeList.createDiv({ cls: "media-home-empty", text: "当前没有进行中的作品。" });
        }
      }, 180);
      return;
    }

    const next = state.known
      ? Math.min(state.total, state.current + state.step)
      : state.current + state.step;
    if (!await saveProgress({ [state.field]: next })) return;
    state.current = next;
    syncControls();
  });

  syncControls();
}
