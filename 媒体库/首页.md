const progressFor = page => {

  const config = domain.progressConfigs[type];

  if (!config) return null;
  const total = config.fixedTotal ?? Math.max(0, Math.round(numberFrom(page[config.totalField])));
  const current = Math.max(0, Math.round(numberFrom(page[config.field])));

    current: known ? Math.min(current, total) : current,
    total,
    known
  };

const ensureFolder = async folderPath => {
  let current = "";
  for (const part of folderPath.split("/").filter(Boolean)) {
    current = current ? `${current}/${part}` : part;
    if (!app.vault.getAbstractFileByPath(current)) {
      try {
        await app.vault.createFolder(current);
      } catch (error) {
        if (!app.vault.getAbstractFileByPath(current)) throw error;
      }
    }
  }
};

const experienceFileFor = (page, index) => app.vault.getMarkdownFiles().find(file => {
  if (!file.path.startsWith("媒体库/记录/")) return false;
  const frontmatter = app.metadataCache.getFileCache(file)?.frontmatter || {};
  if (frontmatter.note_type !== "media_experience" || experienceIndexFrom(frontmatter.experience_index) !== index) return false;
  const target = linkTarget(frontmatter.work);
  const resolved = target ? app.metadataCache.getFirstLinkpathDest(target, file.path) : null;
  return resolved?.path === page.file.path || `${target}.md` === page.file.path || target === page.file.path;
});

const upsertCompletedExperience = async (page, state, startedAt, endedAt) => {
  const index = experienceIndexFrom(page.experience_index);
  const workPath = page.file.path.replace(/\.md$/i, "");
  const label = page.media_type === "book" ? "阅读" : page.media_type === "game" ? "游玩" : "观看";
  const progressType = domain.formatFor(page, typeController);
  const unit = domain.progressConfigs[progressType]?.historyUnit || "percent";
  let file = experienceFileFor(page, index);
  const wasReopened = file && app.metadataCache.getFileCache(file)?.frontmatter?.record_state === "reopened";
  let isNew = false;
  if (!file) {
    const folder = `媒体库/记录/${endedAt.slice(0, 4)}`;
    await ensureFolder(folder);
    const safeName = page.file.name.replace(/[\\/:*?"<>|#\[\]]/g, "-");
    const path = `${folder}/${safeName} · 第${index}次.md`;
    file = app.vault.getAbstractFileByPath(path);
    if (!file) {
      const body = `---\nnote_type: media_experience\nwork: ${JSON.stringify(`[[${workPath}|${mediaTitle(page)}]]`)}\nexperience_index: ${index}\nresult: completed\nrecord_origin: tracked\ndate_certainty: exact\nyear_verified: true\nverified_year: ${endedAt.slice(0, 4)}\nstarted_at: ${startedAt || ""}\nended_at: ${endedAt}\nprogress_value: ${state.current}\nprogress_total: ${state.total || ""}\nprogress_unit: ${unit}\nexperience_rating: ${numberFrom(page.rating) > 0 ? numberFrom(page.rating) : "null"}\nedition: ${page.edition ? JSON.stringify(String(page.edition)) : ""}\nplayed_on: []\ncreated_at: ${localToday()}\ncssclasses:\n  - media-library-page\n---\n\n# ${mediaTitle(page)} · 第 ${index} 次${label}\n\n[[${workPath}|← 返回作品]]\n\n## 本次记录\n\n\`\`\`dataviewjs\nawait dv.view("媒体库/视图/体验记录")\n\`\`\`\n\n`;
      file = await app.vault.create(path, body);
      isNew = true;
    }
  }
  await domain.runFileWrite(app.vault.getName(), file.path, () => app.fileManager.processFrontMatter(file, frontmatter => {
    frontmatter.note_type = "media_experience";
    frontmatter.work = `[[${workPath}|${mediaTitle(page)}]]`;
    frontmatter.experience_index = index;
    frontmatter.result = "completed";
    frontmatter.record_state = "ended";
    if (!frontmatter.record_origin) frontmatter.record_origin = "tracked";
    if (isNew || wasReopened) {
      frontmatter.date_certainty = "exact";
      frontmatter.year_verified = true;
      frontmatter.verified_year = Number(endedAt.slice(0, 4));
      frontmatter.started_at = startedAt || null;
      frontmatter.ended_at = endedAt;
    }
    frontmatter.progress_value = state.current;
    frontmatter.progress_total = state.total || null;
    frontmatter.progress_unit = unit;
    frontmatter.experience_rating = numberFrom(page.rating) > 0 ? numberFrom(page.rating) : null;
    if (!Array.isArray(frontmatter.played_on)) frontmatter.played_on = [];
    if (!frontmatter.created_at) frontmatter.created_at = localToday();
  }));
};

  const row = activeList.createDiv({
    cls: `media-home-active-item is-${page.media_type || "media"}`,
    attr: { role: "link", tabindex: "0", "aria-label": `打开 ${mediaTitle(page)}` }
  });
  const openPage = () => app.workspace.openLinkText(page.file.path, "", false);
  row.addEventListener("click", event => {
    if (!event.target.closest("button")) openPage();
  });
  row.addEventListener("keydown", event => {
    if (event.target !== row || (event.key !== "Enter" && event.key !== " ")) return;
    event.preventDefault();
    openPage();
  });

  const thumb = row.createDiv({ cls: `media-home-active-cover is-${page.media_type || "media"}` });
  const cover = coverUrl(page);
  if (cover) {
    const image = thumb.createEl("img");
    image.setAttr("src", cover);
    image.setAttr("alt", "");
    image.setAttr("loading", "lazy");
  } else {
    thumb.createSpan({ text: typeMeta[page.media_type]?.label || "作品" });
  }

  const copy = row.createDiv({ cls: "media-home-active-copy" });
  copy.createDiv({ cls: "media-home-active-title", text: mediaTitle(page) });
  const detail = copy.createDiv({ cls: "media-home-active-detail" });
  detail.createSpan({
    cls: `media-home-active-type is-${page.media_type || "media"}`,
    text: typeMeta[page.media_type]?.label || "作品"
  });
  if (typeof page.rating === "number" && page.rating > 0) {
    detail.createSpan({ text: `${page.rating} ★` });
  }

  const state = progressFor(page);
  if (!state) continue;
  let hasStartedDate = Boolean(page.started_at);
  let hasFinishedDate = Boolean(page.finished_at);

  const progressRow = copy.createDiv({ cls: "media-home-progress-row" });
  const track = progressRow.createDiv({ cls: "media-home-progress-track" });
  const fill = track.createDiv({ cls: "media-home-progress-fill" });
  const controls = progressRow.createDiv({ cls: "media-home-progress-controls" });
  const decrease = controls.createEl("button", {
    cls: "media-home-progress-button",
    attr: { type: "button", "aria-label": `减少《${mediaTitle(page)}》的进度` }
  });
  setAppIcon(decrease, "minus", "−");
  const valueButton = controls.createEl("button", {
    cls: "media-home-progress-value",
    attr: { type: "button", "aria-label": `编辑《${mediaTitle(page)}》的进度` }
  });
  const action = controls.createEl("button", {
    cls: "media-home-progress-button media-home-progress-action",
    attr: { type: "button" }
  });

  const isComplete = () => state.known && state.current >= state.total;
  const syncControls = () => {
    const complete = isComplete();
    const label = progressLabel(state);
    track.toggleClass("is-unknown", !state.known);
    fill.style.width = `${progressPercent(state)}%`;
    valueButton.setText(label);
    valueButton.toggleClass("is-long", label.length > 8);
    valueButton.setAttr("title", label);
    valueButton.setAttr("aria-label", `编辑《${mediaTitle(page)}》的进度，当前为 ${label}`);
    decrease.disabled = state.current <= 0;
    action.toggleClass("is-complete", complete);
    action.setAttr("aria-label", complete
      ? `将《${mediaTitle(page)}》标记为已完成`
      : `增加《${mediaTitle(page)}》的进度`);
    if (complete) action.setText("完成");
    else setAppIcon(action, "plus", "+");
  };

  const setBusy = busy => {
    row.toggleClass("is-updating", busy);
    decrease.disabled = busy || state.current <= 0;
    valueButton.disabled = busy;
    action.disabled = busy;
  };

  const saveProgress = async values => {
    setBusy(true);
    try {
      const updates = {
        ...values,
        experience_index: experienceIndexFrom(page.experience_index)
      };
      if (Object.prototype.hasOwnProperty.call(values, state.field)) updates.last_activity_at = localToday();
      await updateFrontmatter(page, updates);
      return true;
    } catch (error) {
      console.error(error);
      showProgressNotice("进度更新失败，请稍后重试。");
      return false;
    } finally {
      setBusy(false);
    }
  };

  const addMissingStartDate = values => hasStartedDate
    ? values
    : { ...values, started_at: localToday() };

  decrease.addEventListener("click", async event => {
    event.preventDefault();
    event.stopPropagation();
    const next = Math.max(0, state.current - state.step);
    if (next === state.current || !await saveProgress(addMissingStartDate({ [state.field]: next }))) return;
    hasStartedDate = true;
    state.current = next;
    syncControls();
  });

  valueButton.addEventListener("click", async event => {
    event.preventDefault();
    event.stopPropagation();
    const example = state.unit === "%"
      ? "例如：65"
      : `可以输入当前${state.unit}数，或输入“当前/总${state.unit}数”`;
    const input = window.prompt(`编辑《${mediaTitle(page)}》进度\n${example}`, state.known
      ? `${state.current}/${state.total}`
      : String(state.current));
    if (input === null) return;

    const rawParts = input.split(/[\/／]/);
    const parts = rawParts.map(progressInputNumber);
    if (parts.length > 2 || parts.some(value => value === null)) {
      showProgressNotice("请输入有效的进度数字。");
      return;
    }
    const nextCurrent = Math.max(0, Math.round(parts[0]));

    const values = { [state.field]: nextCurrent };
    let nextTotal = state.total;
    if (state.totalField && parts.length > 1) {
      nextTotal = Math.max(0, Math.round(parts[1]));
      values[state.totalField] = nextTotal;
    }
    const nextKnown = state.fixedTotal ? true : nextTotal > 0;
    const clampedCurrent = nextKnown ? Math.min(nextCurrent, nextTotal) : nextCurrent;
    values[state.field] = clampedCurrent;
    if (!await saveProgress(addMissingStartDate(values))) return;
    hasStartedDate = true;
    state.current = clampedCurrent;
    state.total = nextTotal;
    state.known = nextKnown;
    syncControls();
  });

  action.addEventListener("click", async event => {
    event.preventDefault();
    event.stopPropagation();
    if (isComplete()) {
      const today = localToday();
      const startedAt = hasStartedDate ? dateText(page.started_at) : today;
      const finishedAt = hasFinishedDate ? dateText(page.finished_at) : today;
      const completionValues = {
        status: "已完成",
        experience_index: experienceIndexFrom(page.experience_index),
        last_activity_at: today
      };
      if (!hasStartedDate) completionValues.started_at = today;
      if (!hasFinishedDate) completionValues.finished_at = today;
      if (!await saveProgress(completionValues)) return;
      hasStartedDate = true;
      hasFinishedDate = true;
      try {
        await upsertCompletedExperience(page, state, startedAt, finishedAt);
      } catch (error) {
        console.error("首页体验历史保存失败", error);
        showProgressNotice("作品已完成，但体验历史保存失败；可在作品详情页重试。");
      }
      row.addClass("is-finished");
      window.setTimeout(() => {
        row.remove();
        if (!activeList.querySelector(".media-home-active-item")) {
          activeList.createDiv({ cls: "media-home-empty", text: "当前没有进行中的作品。" });
        }
      }, 180);
      return;
    }

    const next = state.known
      ? Math.min(state.total, state.current + state.step)
      : state.current + state.step;
    if (!await saveProgress(addMissingStartDate({ [state.field]: next }))) return;
    hasStartedDate = true;
    state.current = next;
    syncControls();
  });

  syncControls();
}

// 待体验：接在最近加入下方，与右侧进行中区域共同留在首屏。
const backlogSection = libraryColumn.createDiv({ cls: "media-home-section media-home-backlog" });
let backlogItems = items
  .filter(page => page.status === "待体验")
  .sort((a, b) => timestamp(b.added_at) - timestamp(a.added_at)
    || timestamp(b.file?.mtime) - timestamp(a.file?.mtime));
let backlogBatch = backlogItems.slice(0, 5);

const shuffled = values => {
  const result = [...values];
  for (let index = result.length - 1; index > 0; index -= 1) {
    const target = Math.floor(Math.random() * (index + 1));
    [result[index], result[target]] = [result[target], result[index]];
  }
  return result;
};

const nextBacklogBatch = () => {
  const currentPaths = new Set(backlogBatch.map(page => page.file.path));
  const unseen = backlogItems.filter(page => !currentPaths.has(page.file.path));
  const previous = backlogItems.filter(page => currentPaths.has(page.file.path));
  backlogBatch = [...shuffled(unseen), ...shuffled(previous)].slice(0, 5);
};

const backlogDateLabel = page => {
  const addedAt = dateText(page.added_at);
  if (!addedAt) return "加入时间未知";
  const year = Number(addedAt.slice(0, 4));
  const month = Number(addedAt.slice(5, 7));
  return year === new Date().getFullYear() ? `${month} 月加入` : `${year} 年加入`;
};
